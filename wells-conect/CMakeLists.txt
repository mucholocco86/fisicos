cmake_minimum_required(VERSION 3.21)
project(WellsConect VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network)
qt_add_executable(WellsConect WIN32 main.cpp mainwindow.h mainwindow.cpp)
target_link_libraries(WellsConect PRIVATE Qt6::Widgets Qt6::Network)
set_target_properties(WellsConect PROPERTIES WIN32_EXECUTABLE TRUE)
