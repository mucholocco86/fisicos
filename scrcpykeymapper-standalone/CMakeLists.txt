cmake_minimum_required(VERSION 3.21)
project(ScrcpyKeyMapperStandalone VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets WebEngineWidgets Network)
qt_standard_project_setup()
qt_add_executable(ScrcpyKeyMapper WIN32 main.cpp)
target_link_libraries(ScrcpyKeyMapper PRIVATE Qt6::Widgets Qt6::WebEngineWidgets Qt6::Network)
install(TARGETS ScrcpyKeyMapper RUNTIME DESTINATION .)
